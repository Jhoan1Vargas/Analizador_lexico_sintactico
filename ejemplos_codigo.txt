═══════════════════════════════════════════════════════════════════════════
  NUEVOS TOKENS SOPORTADOS
═══════════════════════════════════════════════════════════════════════════

PALABRAS RESERVADAS:
  Control: if, else, while, for, do, break, continue, return
  Tipos: int, float, double, char, void, bool
  Booleanos: true, false

OPERADORES:
  Aritméticos: +, -, *, /, %
  Relacionales: ==, !=, <, >, <=, >=
  Lógicos: &&, ||, !
  Asignación: =, +=, -=, *=, /=
  Inc/Dec: ++, --

NÚMEROS:
  Enteros: 42, 100, 0
  Decimales: 3.14, 2.718, 0.5

DELIMITADORES:
  Paréntesis: ( )
  Llaves: { }
  Corchetes: [ ]
  Otros: ; , .

IDENTIFICADORES:
  Formato: letra/_ seguido de letras/dígitos/_
  Ejemplos: x, miVariable, _privado, var_123

COMENTARIOS:
  Línea: // comentario
  Bloque: /* comentario */
